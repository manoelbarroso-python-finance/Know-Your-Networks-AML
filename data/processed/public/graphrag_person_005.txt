CYPHER GERADO PELO LLM
----------------------
MATCH (p:Person {id: 'PERSON_005'})

CALL {
    WITH p
    RETURN {
        id: p.id,
        nome: p.nome,
        pais: p.pais,
        segmento: p.segmento
    } AS dados_basicos
}

CALL {
    WITH p
    OPTIONAL MATCH (p)-[:MATCHED_TO]->(r:RiskEntity)
    RETURN collect(DISTINCT {
        id: r.id,
        nome: r.nome,
        fonte: r.fonte,
        categoria: r.categoria
    }) AS entidades_risco
}

CALL {
    WITH p
    MATCH (p)-[:USES_DEVICE]->(d:Device)
    OPTIONAL MATCH (outra:Person)-[:USES_DEVICE]->(d)
    WHERE outra <> p
    WITH d, collect(DISTINCT outra.id) AS outras_pessoas
    RETURN collect({
        id: d.id,
        tipo: d.tipo,
        outras_pessoas: outras_pessoas
    }) AS dispositivos
}

CALL {
    WITH p
    OPTIONAL MATCH (p)-[:OWNS]->(a:Account)
    RETURN collect(DISTINCT {
        id: a.id,
        tipo: a.tipo,
        moeda: a.moeda
    }) AS contas
}

CALL {
    WITH p
    MATCH (p)-[:OWNS]->(a:Account)
    MATCH (a)-[t:TRANSFERRED_TO]-(contraparte:Account)
    WHERE t.timestamp >= datetime('2026-07-20T00:00:00Z')
      AND t.timestamp < datetime('2026-07-21T00:00:00Z')
    RETURN collect(DISTINCT {
        conta: a.id,
        direcao: CASE
            WHEN startNode(t) = a THEN 'enviada'
            ELSE 'recebida'
        END,
        contraparte_conta: contraparte.id,
        contraparte_proprietarios: [
            (proprietario)-[:OWNS]->(contraparte) | proprietario.id
        ],
        valor: t.valor,
        moeda: t.moeda,
        timestamp: t.timestamp,
        tx_id: t.tx_id
    }) AS transferencias
}

RETURN
    dados_basicos,
    entidades_risco,
    dispositivos,
    contas,
    transferencias
